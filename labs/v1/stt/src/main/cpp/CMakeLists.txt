cmake_minimum_required(VERSION 3.22.1)
project(jarvis_stt LANGUAGES C CXX)
# whisper.cpp pinned source (tools/prepare_whisper.sh, commit 9386f239 = v1.8.4), CPU only, static ggml
# kept PRIVATE to this library: only Java_fr_jarvis_v1_stt_* is exported, so it can never interpose
# with the ggml embedded in the A02 LLM library (different pinned version).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
    message(FATAL_ERROR "Run labs/v1/tools/prepare_whisper.sh first (WHISPER_SOURCE_DIR)")
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" "whisper-build" EXCLUDE_FROM_ALL)
add_library(jarvis_stt SHARED whisper_jni.cpp)
target_link_libraries(jarvis_stt PRIVATE whisper)
target_link_options(jarvis_stt PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,-Bsymbolic" "-Wl,--exclude-libs,ALL")
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(jarvis_stt PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(ANDROID)
    target_link_libraries(jarvis_stt PRIVATE android log)
    target_link_options(jarvis_stt PRIVATE "-Wl,-z,max-page-size=16384")
endif()
